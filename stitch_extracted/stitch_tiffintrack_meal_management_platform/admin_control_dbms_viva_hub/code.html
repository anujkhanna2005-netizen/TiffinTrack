CREATE OR REPLACE TRIGGER trg_auto_meal_refund
AFTER UPDATE OF status ON daily_deliveries
FOR EACH ROW
WHEN (NEW.status = 'SKIPPED' AND NEW.requested_before_cutoff = TRUE)
BEGIN
-- 1. Automate instant campus wallet reimbursement (No admin intervention needed)
UPDATE student_wallets
SET balance = balance + 80.00, updated_at = NOW()
WHERE student_id = :NEW.student_id;
-- 2. Append irreversible audit event to WAL ledger
INSERT INTO wallet_audit_ledger (student_id, amount, reason, txn_type)
VALUES (:NEW.student_id, 80.00, 'Meal Skip Credit: TT-LUNCH', 'CREDIT');
END;
